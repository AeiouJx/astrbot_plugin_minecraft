# 调用 AI

本节介绍如何在 AstrBot 插件中调用 AI 功能，包括获取聊天提供者、调用 LLM、定义和注册工具，以及使用 Agent。

## 获取聊天提供者 ID

### 获取默认提供者

```python
from astrbot.api.star import Context

class AIPlugin(Star):
    def __init__(self, context: Context):
        super().__init__(context)
        
    async def get_chat_provider(self):
        """获取默认聊天提供者"""
        provider = await self.context.get_default_chat_provider()
        return provider
```

### 获取指定提供者

```python
async def get_specific_provider(self, provider_id: str):
    """获取指定的聊天提供者"""
    provider = await self.context.get_chat_provider(provider_id)
    return provider
```

### 列出所有提供者

```python
async def list_providers(self):
    """列出所有可用的聊天提供者"""
    providers = await self.context.list_chat_providers()
    for provider in providers:
        print(f"提供者: {provider.id}, 类型: {provider.type}")
    return providers
```

## 调用 LLM

### 基本 LLM 调用

```python
@filter.command("ask")
async def on_ask(self, event: AstrMessageEvent):
    """调用 LLM 回答问题"""
    question = event.message_str.replace("/ask", "").strip()
    if not question:
        await event.plain_result("请输入问题，格式: /ask <问题>")
        return
        
    try:
        # 获取聊天提供者
        provider = await self.context.get_default_chat_provider()
        
        # 调用 LLM
        response = await provider.llm_generate(
            messages=[{"role": "user", "content": question}]
        )
        
        await event.plain_result(response)
    except Exception as e:
        await event.plain_result(f"调用 LLM 失败: {str(e)}")
```

### 带系统提示的调用

```python
async def call_llm_with_system_prompt(self, user_message: str, system_prompt: str):
    """带系统提示的 LLM 调用"""
    provider = await self.context.get_default_chat_provider()
    
    messages = [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": user_message}
    ]
    
    response = await provider.llm_generate(messages=messages)
    return response
```

### 流式调用

```python
async def stream_llm_response(self, event: AstrMessageEvent, question: str):
    """流式调用 LLM"""
    provider = await self.context.get_default_chat_provider()
    
    messages = [{"role": "user", "content": question}]
    
    # 流式获取响应
    full_response = ""
    async for chunk in provider.llm_generate_stream(messages=messages):
        full_response += chunk
        # 可以在这里实现流式显示
        
    return full_response
```

## 定义工具

### 使用 dataclass 定义工具

```python
from dataclasses import dataclass
from typing import Optional
from astrbot.api.tool import FunctionTool

@dataclass
class WeatherTool(FunctionTool):
    """天气查询工具"""
    
    name: str = "weather"
    description: str = "查询指定城市的天气信息"
    
    async def execute(self, city: str, unit: str = "celsius") -> str:
        """
        查询天气
        
        Args:
            city: 城市名称
            unit: 温度单位 (celsius/fahrenheit)
            
        Returns:
            天气信息字符串
        """
        # 实现天气查询逻辑
        # 这里只是示例，实际应该调用天气 API
        weather_data = {
            "北京": "晴天，温度 25°C",
            "上海": "多云，温度 22°C",
            "广州": "雨天，温度 28°C"
        }
        
        weather = weather_data.get(city, f"未找到 {city} 的天气信息")
        return f"{city} 天气: {weather}"
```

### 定义复杂工具

```python
@dataclass
class DatabaseTool(FunctionTool):
    """数据库查询工具"""
    
    name: str = "database_query"
    description: str = "执行数据库查询"
    
    def __init__(self, db_connection):
        self.db_connection = db_connection
        
    async def execute(self, query: str, params: Optional[dict] = None) -> str:
        """
        执行数据库查询
        
        Args:
            query: SQL 查询语句
            params: 查询参数
            
        Returns:
            查询结果
        """
        try:
            if params:
                result = await self.db_connection.execute(query, params)
            else:
                result = await self.db_connection.execute(query)
                
            return f"查询成功，返回 {len(result)} 条记录"
        except Exception as e:
            return f"查询失败: {str(e)}"
```

## 注册工具

### 使用 add_llm_tools

```python
class ToolPlugin(Star):
    def __init__(self, context: Context):
        super().__init__(context)
        
    @filter.on_astrbot_loaded()
    async def on_loaded(self):
        """注册工具"""
        # 注册天气工具
        weather_tool = WeatherTool()
        await self.context.add_llm_tools([weather_tool])
        
        # 注册数据库工具
        db_tool = DatabaseTool(self.db_connection)
        await self.context.add_llm_tools([db_tool])
```

### 使用 @filter.llm_tool 装饰器

```python
@filter.llm_tool("calculator")
async def calculator_tool(self, expression: str) -> str:
    """
    计算数学表达式
    
    Args:
        expression: 数学表达式，如 "2 + 3 * 4"
        
    Returns:
        计算结果
    """
    try:
        # 安全计算表达式
        result = eval(expression, {"__builtins__": {}}, {})
        return f"计算结果: {result}"
    except Exception as e:
        return f"计算错误: {str(e)}"
        
@filter.llm_tool("translate")
async def translate_tool(self, text: str, target_lang: str = "en") -> str:
    """
    翻译文本
    
    Args:
        text: 要翻译的文本
        target_lang: 目标语言代码
        
    Returns:
        翻译结果
    """
    # 实现翻译逻辑
    # 这里只是示例
    translations = {
        "en": f"Translated: {text}",
        "ja": f"翻訳: {text}",
        "ko": f"번역: {text}"
    }
    return translations.get(target_lang, f"翻译为 {target_lang}: {text}")
```

## 调用 Agent

### 基本 Agent 调用

```python
@filter.command("agent")
async def on_agent(self, event: AstrMessageEvent):
    """调用 Agent 执行任务"""
    task = event.message_str.replace("/agent", "").strip()
    if not task:
        await event.plain_result("请输入任务描述，格式: /agent <任务>")
        return
        
    try:
        # 获取聊天提供者
        provider = await self.context.get_default_chat_provider()
        
        # 调用 Agent
        result = await provider.tool_loop_agent(
            task=task,
            tools=self.get_registered_tools()
        )
        
        await event.plain_result(f"Agent 执行结果:\n{result}")
    except Exception as e:
        await event.plain_result(f"Agent 执行失败: {str(e)}")
```

### 配置 Agent 参数

```python
async def call_agent_with_config(self, task: str, config: dict):
    """带配置的 Agent 调用"""
    provider = await self.context.get_default_chat_provider()
    
    # 配置 Agent
    agent_config = {
        "max_iterations": config.get("max_iterations", 10),
        "temperature": config.get("temperature", 0.7),
        "verbose": config.get("verbose", True)
    }
    
    result = await provider.tool_loop_agent(
        task=task,
        tools=self.get_registered_tools(),
        config=agent_config
    )
    
    return result
```

## Multi-Agent 模式

### 定义多个 Agent

```python
class MultiAgentPlugin(Star):
    def __init__(self, context: Context):
        super().__init__(context)
        self.agents = {}
        
    @filter.on_astrbot_loaded()
    async def on_loaded(self):
        """初始化多个 Agent"""
        # 创建研究 Agent
        self.agents["researcher"] = await self.create_agent(
            name="researcher",
            system_prompt="你是一个研究助手，负责收集和分析信息。",
            tools=[self.search_tool, self.analyze_tool]
        )
        
        # 创建写作 Agent
        self.agents["writer"] = await self.create_agent(
            name="writer",
            system_prompt="你是一个写作助手，负责将信息整理成文章。",
            tools=[self.write_tool, self.edit_tool]
        )
        
    async def create_agent(self, name: str, system_prompt: str, tools: list):
        """创建 Agent"""
        provider = await self.context.get_default_chat_provider()
        
        agent = await provider.create_agent(
            name=name,
            system_prompt=system_prompt,
            tools=tools
        )
        
        return agent
```

### 协作执行

```python
async def collaborative_execution(self, task: str):
    """多 Agent 协作执行"""
    # 第一步：研究
    research_result = await self.agents["researcher"].execute(
        f"研究以下任务: {task}"
    )
    
    # 第二步：写作
    article = await self.agents["writer"].execute(
        f"根据以下研究结果写一篇文章:\n{research_result}"
    )
    
    return article
```

## 会话管理器

### 创建会话

```python
class ConversationManager:
    def __init__(self, context: Context):
        self.context = context
        self.conversations = {}
        
    async def create_conversation(self, user_id: str):
        """创建新会话"""
        conversation = await self.context.create_conversation(
            user_id=user_id,
            metadata={"created_at": "2024-01-01"}
        )
        self.conversations[user_id] = conversation
        return conversation
```

### 管理会话历史

```python
async def add_message_to_history(self, user_id: str, role: str, content: str):
    """添加消息到历史记录"""
    if user_id not in self.conversations:
        await self.create_conversation(user_id)
        
    conversation = self.conversations[user_id]
    await conversation.add_message(role=role, content=content)
    
async def get_conversation_history(self, user_id: str):
    """获取会话历史"""
    if user_id not in self.conversations:
        return []
        
    conversation = self.conversations[user_id]
    return await conversation.get_history()
```

### 清理会话

```python
async def clear_conversation(self, user_id: str):
    """清理会话"""
    if user_id in self.conversations:
        await self.conversations[user_id].clear()
        del self.conversations[user_id]
```

class PersonaManager:

### 使用人格

```python
async def chat_with_persona(self, user_id: str, message: str, persona_name: str):
    """使用指定人格聊天"""
    if persona_name not in self.personas:
        return "未找到指定的人格"
        
    persona = self.personas[persona_name]
    
    # 构建带人格的提示
    system_prompt = f"""
    {persona['system_prompt']}
    
    人格特点: {persona['personality']}
    回复风格: {persona['response_style']}
    """
    
    provider = await self.context.get_default_chat_provider()
    
    messages = [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": message}
    ]
    
    response = await provider.llm_generate(messages=messages)
    return response
```

## 完整 AI 示例

```python
from astrbot.api.star import Context
from astrbot.api.event import filter, AstrMessageEvent
from astrbot.core.star.star import Star
from dataclasses import dataclass
from typing import Optional

@dataclass
class SearchTool:
    """搜索工具"""
    name: str = "search"
    description: str = "搜索互联网获取信息"
    
    async def execute(self, query: str) -> str:
        """执行搜索"""
        # 实现搜索逻辑
        return f"搜索结果: 关于 '{query}' 的信息..."

@dataclass
class CalculatorTool:
    """计算器工具"""
    name: str = "calculator"
    description: str = "计算数学表达式"
    
    async def execute(self, expression: str) -> str:
        """计算表达式"""
        try:
            result = eval(expression, {"__builtins__": {}}, {})
            return f"计算结果: {result}"
        except Exception as e:
            return f"计算错误: {str(e)}"

class AIAssistantPlugin(Star):
    def __init__(self, context: Context):
        super().__init__(context)
        self.tools = [SearchTool(), CalculatorTool()]
        self.conversations = {}
        
    @filter.on_astrbot_loaded()
    async def on_loaded(self):
        """注册工具"""
        await self.context.add_llm_tools(self.tools)
        
    @filter.command("chat")
    async def on_chat(self, event: AstrMessageEvent):
        """与 AI 聊天"""
        message = event.message_str.replace("/chat", "").strip()
        if not message:
            await event.plain_result("请输入消息，格式: /chat <消息>")
            return
            
        try:
            # 获取或创建会话
            user_id = event.sender.id
            if user_id not in self.conversations:
                self.conversations[user_id] = []
                
            # 添加用户消息到历史
            self.conversations[user_id].append({
                "role": "user",
                "content": message
            })
            
            # 调用 LLM
            provider = await self.context.get_default_chat_provider()
            response = await provider.llm_generate(
                messages=self.conversations[user_id]
            )
            
            # 添加 AI 回复到历史
            self.conversations[user_id].append({
                "role": "assistant",
                "content": response
            })
            
            # 限制历史长度
            if len(self.conversations[user_id]) > 20:
                self.conversations[user_id] = self.conversations[user_id][-20:]
                
            await event.plain_result(response)
        except Exception as e:
            await event.plain_result(f"聊天失败: {str(e)}")
            
    @filter.command("agent_task")
    async def on_agent_task(self, event: AstrMessageEvent):
        """使用 Agent 执行任务"""
        task = event.message_str.replace("/agent_task", "").strip()
        if not task:
            await event.plain_result("请输入任务描述，格式: /agent_task <任务>")
            return
            
        try:
            provider = await self.context.get_default_chat_provider()
            
            # 调用 Agent
            result = await provider.tool_loop_agent(
                task=task,
                tools=self.tools,
                config={"max_iterations": 5}
            )
            
            await event.plain_result(f"Agent 执行完成:\n{result}")
        except Exception as e:
            await event.plain_result(f"Agent 执行失败: {str(e)}")
            
    @filter.command("persona")
    async def on_persona(self, event: AstrMessageEvent):
        """设置 AI 人格"""
        args = event.message_str.split()[1:]
        if not args:
            await event.plain_result("可用人格: formal, casual, friendly")
            return
            
        persona_name = args[0]
        personas = {
            "formal": "你是一个正式的助手，使用专业、礼貌的语言。",
            "casual": "你是一个轻松的助手，使用随意、友好的语言。",
            "friendly": "你是一个热情的助手，使用温暖、亲切的语言。"
        }
        
        if persona_name in personas:
            # 这里可以保存用户的人格设置
            await event.plain_result(f"已切换到 {persona_name} 人格")
        else:
            await event.plain_result("未找到指定的人格")
```
